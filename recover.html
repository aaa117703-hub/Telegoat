استرجاع نتائج Telegoat

الخطوة 1: أنشئ جدول Telegoat الخاص (مرة واحدة) — انسخ والصق في Supabase ← SQL Editor ← Run: 
create table if not exists public.tg_match_results (
  id bigint generated always as identity primary key,
  round text not null,
  home_team text not null,
  away_team text not null,
  home_score int,
  away_score int,
  created_at timestamptz default now()
);
alter table public.tg_match_results enable row level security;
drop policy if exists tg_mr_all on public.tg_match_results;
create policy tg_mr_all on public.tg_match_results for all using (true) with check (true);
